34. def intersection(a, b):
    res = []
    for x in a:
        if x in b and x not in res:
            res.append(x)
    return res

35. def flatten(lst):
    res = []
    for i in lst:
        if isinstance(i, list):
            res.extend(i)
        else:
            res.append(i)
    return res

37. import sqlite3
DB_PATH = 'students.db'

def add_student(name, grade):
    conn = sqlite3.connect(DB_PATH)
    try:
        conn.execute("INSERT INTO students (name, grade) VALUES (?, ?)", (name, grade))
        conn.commit()
        return True
    except sqlite3.Error:
        conn.rollback()
        return False
    finally:
        conn.close()

def get_all_students():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    try:
        return [tuple(r) for r in conn.execute("SELECT id, name, grade FROM students")]
    except sqlite3.Error:
        return []
    finally:
        conn.close()

def update_grade(sid, grade):
    conn = sqlite3.connect(DB_PATH)
    try:
        cur = conn.execute("UPDATE students SET grade = ? WHERE id = ?", (grade, sid))
        conn.commit()
        return cur.rowcount
    except sqlite3.Error:
        conn.rollback()
        return 0
    finally:
        conn.close()

38. from flask import Flask, render_template, request, redirect, url_for
import sqlite3

app = Flask(__name__)

@app.route('/students', methods=['GET', 'POST'])
def students():
    if request.method == 'POST':
        name = request.form.get('name', '').strip()
        grade = request.form.get('grade', '')
        if not name or not grade.isdigit():
            return render_template('students.html', students=[], error='Ошибка')
        conn = sqlite3.connect('app.db')
        try:
            conn.execute("INSERT INTO students (name, grade) VALUES (?, ?)", (name, int(grade)))
            conn.commit()
        except sqlite3.Error as e:
            conn.rollback()
            return render_template('students.html', students=[], error=str(e))
        finally:
            conn.close()
        return redirect(url_for('students'))
    conn = sqlite3.connect('app.db')
    conn.row_factory = sqlite3.Row
    try:
        s = conn.execute("SELECT * FROM students").fetchall()
    except:
        s = []
    finally:
        conn.close()
    return render_template('students.html', students=s)

35. def stats(students):
    if not students:
        return (None, None, None)
    g = [x[1] for x in students]
    return (min(g), max(g), round(sum(g)/len(g), 2))

36. def unique_words(text):
    res = []
    for w in text.split():
        w = w.lower()
        if w not in res:
            res.append(w)
    return res

39. def top_authors(books, n):
    counts = {}
    for _, author in books:
        counts[author] = counts.get(author, 0) + 1
    items = list(counts.items())
    for i in range(len(items)):
        for j in range(len(items)-1-i):
            if items[j][1] < items[j+1][1] or (items[j][1] == items[j+1][1] and items[j][0] > items[j+1][0]):
                items[j], items[j+1] = items[j+1], items[j]
    return [x[0] for x in items[:n]]
